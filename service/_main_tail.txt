@app.post("/analyze")
async def analyze(file: UploadFile = File(...)):
    t0 = time.time()
    bgr = _decode(await file.read())
    full, mask = _flame_mask(bgr)
    temp = _temp_index(bgr)
    spots = _wall_hotspots(temp, full, mask)

    tag = f"{int(time.time())}_{uuid.uuid4().hex[:6]}"
    contour_name, heat_name = f"{tag}_contour.jpg", f"{tag}_heat.jpg"
    q = [cv2.IMWRITE_JPEG_QUALITY, 90]
    cv2.imwrite(os.path.join(STATIC_DIR, contour_name), _draw_contours(bgr, temp, mask, spots), q)
    # 热力图用 full：贴壁亮斑也要显示成高温，不能被压暗
    cv2.imwrite(os.path.join(STATIC_DIR, heat_name), _draw_heatmap(bgr, temp, full), q)

    return {
        "metrics": _metrics(bgr, temp, mask, full, spots),
        "contour_url": f"{BASE_URL}/static/{contour_name}",
        "heatmap_url": f"{BASE_URL}/static/{heat_name}",
        "elapsed_ms": int((time.time() - t0) * 1000),
    }


@app.post("/report")
async def make_report(
    assessment_text: str = Form(""),
    metrics_json: str = Form("{}"),
    contour_url: str = Form(""),
    heatmap_url: str = Form(""),
    shot_at: str = Form(""),
    site: str = Form(""),
):
    """
    把模型判级 + CV 指标渲染成一张独立 HTML 报告页，返回它的地址。

    走 form-data 而不是 JSON：模型输出里带换行和引号，塞进 Dify 的 JSON body
    模板会把请求体撑坏，表单字段没这个问题。
    """
    try:
        metrics = json.loads(metrics_json) if metrics_json.strip() else {}
        if not isinstance(metrics, dict):
            metrics = {}
    except Exception:
        metrics = {}

    if not shot_at:
        shot_at = time.strftime("%Y-%m-%d %H:%M")

    # 报告页文件名要先定下来，markdown 开头得引用它
    name = f"{int(time.time())}_{uuid.uuid4().hex[:6]}_report.html"
    report_url = f"{BASE_URL}/static/{name}"

    # 两套模板共用一份上下文，模型输出只解析一次
    ctx = report_tpl.build_context(
        assessment_text, metrics, contour_url, heatmap_url,
        report_url, shot_at, site
    )
    with open(os.path.join(STATIC_DIR, name), "w", encoding="utf-8") as f:
        f.write(report_tpl.render_html(ctx))

    return {
        "report_url": report_url,
        "risk_level": ctx["level"],
        "markdown": report_tpl.render_markdown_from(ctx),
    }
